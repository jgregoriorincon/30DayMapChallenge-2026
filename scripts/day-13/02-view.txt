# Día 13 — Interactions: El Magdalena en ocho paradas — 2 · VISTA (estilo, fichas y marcadores)

# Orden: paradas arriba, el río, la cuenca abajo. layer.move de una sola capa falla si ya está en
# ese puesto, y el orden de llegada no se conoce (día 12), así que:
# 1 · Paradas y río arriba: la cuenca queda seguro al fondo (puesto 2)
layer.move {"layer": ["Paradas", "Río Magdalena"], "to": 0}
# 2 · Paradas y cuenca arriba (conservan su orden: paradas, cuenca): el río queda al fondo
layer.move {"layer": ["Paradas", "Cuenca Magdalena-Cauca"], "to": 0}
# 3 · El río al puesto 1 (ahí está la cuenca): paradas, río, cuenca
layer.move {"layer": "Río Magdalena", "to": 1}

# La cuenca: un velo blanco muy suave y sus zonas con borde fino, para que el relieve se siga viendo
layer.color {"layer": "Cuenca Magdalena-Cauca", "color": "#FFFFFF40"}
layer.outline {"layer": "Cuenca Magdalena-Cauca", "color": "#2B5D8A", "width": 0.5}
# El río: azul intenso, con borde del mismo azul para que no se corte donde el cauce es angosto
layer.color {"layer": "Río Magdalena", "color": "#0B5FA5FF"}
layer.outline {"layer": "Río Magdalena", "color": "#0B5FA5", "width": 0.9}
# Paradas: círculo rojo de 36 (9,5 mm en el A3: en RustGIS un "punto" de símbolo es 1/96 de pulgada, 0,265 mm,
# PT_PER_MM = 96 / 25,4 en crates/map/src/layout.rs) con borde blanco y su número encima
layer.marker {"layer": "Paradas", "shape": "circle", "size": 36}
layer.color {"layer": "Paradas", "color": "#C62828"}
layer.outline {"layer": "Paradas", "color": "#FFFFFF", "width": 2.2}
labels.field {"layer": "Paradas", "field": "parada"}
labels.style {"layer": "Paradas", "size": 18, "bold": true, "color": "#FFFFFF", "halo": false}
labels.placement {"layer": "Paradas", "position": "center", "showAll": true}
