# Día 24 — Network: Antes de ser desfile, era un camino — 2 · VISTA
# Correr cuando el MDE del script 1 esté en el panel.
# El gesto: un bloque 3D del relieve de 1 m con el camino en rosa (las flores del día 19) y, aparte, el
# perfil del camino pintado con los mismos colores de altura: se leen como una sola pieza.

map.activate {"name": "Medellín"}

# El TIFF del servidor no declara su "sin dato": una copia limpia con álgebra de bandas. En un píxel de
# -32 768, 1 / (b1 + 32 768) es infinito y 0 × infinito no es un número; bandMath lo guarda como "sin dato"
raster.function {"layer": "MDE Medellín 2024", "function": "bandMath", "expression": "b1 + 0 * (1 / (b1 + 32768))", "name": "MDE limpio"}
layer.remove {"layer": "MDE Medellín 2024"}

# Relieve sombreado con rampa Terrain, estirada a mano de 1 400 a 2 800 m: el perfil usa los mismos colores.
# Sin sombras proyectadas: en el navegador solo se calculan hasta 500 000 celdas y el MDE tiene 8 266 800
raster.symbology {"layer": "MDE limpio", "type": "shadedRelief", "ramp": "Terrain", "azimuth": 315, "altitude": 40}
raster.stretch {"layer": "MDE limpio", "type": "custom", "min": 1400, "max": 2800}

# El camino: rosa con borde blanco; el km más empinado, más grueso y oscuro
layer.lineStrokes {"layer": "Camino", "strokes": [{"color": "#FFFFFF", "width": 4, "cap": "round", "join": "round"}, {"color": "#C2185B", "width": 2.2, "cap": "round", "join": "round"}]}
layer.lineStrokes {"layer": "Tramo empinado", "strokes": [{"color": "#FFFFFF", "width": 5.6, "cap": "round", "join": "round"}, {"color": "#7A0E3A", "width": 3.6, "cap": "round", "join": "round"}]}

# Marcas: puntos rosa con borde blanco y etiquetas con halo
layer.marker {"layer": "Marcas", "shape": "circle", "size": 7}
layer.color {"layer": "Marcas", "color": "#C2185BFF"}
layer.outline {"layer": "Marcas", "color": "#FFFFFF", "width": 1.2}
layer.marker {"layer": "Nota empinado", "shape": "circle", "size": 0.1}
layer.color {"layer": "Nota empinado", "color": "#7A0E3A00"}
layer.outline {"layer": "Nota empinado", "color": "#7A0E3A00", "width": 0}
# Solo las marcas de km: las de los extremos van en el layout (en 3D, Santa Elena no salía y la Placita
# quedaba lejos de su punto)

labels.placement {"layer": "Marcas", "position": "right", "showAll": true}
labels.by {"layer": "Nota empinado", "text": "es"}
labels.style {"layer": "Nota empinado", "size": 12, "bold": true, "color": "#7A0E3A", "halo": true, "haloColor": "#FFFFFF"}
# Más a la izquierda y arriba: su punto está a 50 m del de km 10 y las etiquetas se tocaban
labels.placement {"layer": "Nota empinado", "position": "left", "offset": [-8, -4], "showAll": true}

# Orden: el camino y las marcas sobre el relieve
# Una sola orden con la lista: no falla si alguna ya está arriba (una por una, sí)
layer.move {"layer": ["Camino", "Tramo empinado", "Marcas", "Nota empinado"], "to": 0}

# Bloque 3D solo alrededor del camino (unos 800 m de margen) y dentro de la cobertura del MDE: fuera de
# ella el terreno baja a 0 m y salen cortinas
map.terrain {"layer": "MDE limpio", "exaggeration": 1.5, "block": {"extent": [-8412303, 692092, -8403954, 697691], "depth": 400, "walls": "#B9A88C"}}
# Norte arriba: el lado largo del bloque (este-oeste) horizontal, Santa Elena abajo a la derecha y
# Medellín arriba a la izquierda. Centro 1,5 km al norte del centro del bloque: el relieve exagerado
# sube el bloque en la vista. Inclinación de 35°
map.goToXY {"lon": -75.5292, "lat": 6.2440, "scale": 62500, "bearing": 0, "pitch": 35}
