# Los perímetros llegan con vértices cada pocos metros (unos 900 KB): simplificados a 5 m
# (tolerancia en km), que en el layout es menos de 0,02 mm
gp.run {"tool": "simplify", "params": {"input": "Zonas urbanas", "tolerance": 0.005, "output": "Perímetro urbano · Urban perimeter"}}
layer.remove {"layer": "Zonas urbanas"}

# Cada celda, dentro o fuera del perímetro según dónde cae su centro: join_count = 1 dentro, 0 fuera
gp.run {"tool": "spatialJoin", "params": {"target": "Grilla 1 km", "join": "Perímetro urbano · Urban perimeter", "match": "Have their center in", "output": "Grado de urbanización · Degree of urbanisation"}}
layer.remove {"layer": "Grilla 1 km"}

# La cifra del subtítulo: personas dentro (1) y fuera (0) del perímetro. Debe dar 911 390 y 189 842,
# en 88 y 1142 celdas (FREQUENCY) (réplica en Python, 2026-11-07). Sale una tabla
# "…_Statistics", que no se dibuja
gp.run {"tool": "summaryStatistics", "params": {"input": "Grado de urbanización · Degree of urbanisation", "statistics": [{"field": "personas", "statistic": "SUM"}], "case": "join_count"}}

# La leyenda ordena los valores únicos por número, de menor a mayor: un rango de 1 (ciudad) a 7
# (agua) deja la ciudad arriba. degurba_l2 es texto

edit.calculateField {"layer": "Grado de urbanización · Degree of urbanisation", "field": "Clase", "expression": "CASE WHEN degurba_l2 = '30' THEN 1 WHEN degurba_l2 = '23' THEN 2 WHEN degurba_l2 = '22' THEN 3 WHEN degurba_l2 = '21' THEN 4 WHEN degurba_l2 = '13' THEN 5 WHEN degurba_l2 = '12' THEN 6 WHEN degurba_l2 = '11' THEN 7 ELSE 8 END"}
