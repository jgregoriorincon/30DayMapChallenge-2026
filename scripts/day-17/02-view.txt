# Día 17 — Light & dark: La luz no llega al campo — 2 · VISTA (proceso y estilo)
# Correr cuando "Departamentos" y "Municipios" ya estén en el panel de capas.
# El gesto: dos luces. Las ciudades en ámbar y las viviendas rurales sin servicio encendidas en cian.
# Fuera de Colombia, una máscara casi opaca: los vecinos brillaban más que el país (primera prueba,
# 2026-11-15).

# Un punto por municipio, siempre dentro de su polígono (Feature To Point usa un punto interior, no el
# centroide, que en un municipio con forma de media luna puede caer fuera). Lleva todos los campos
gp.run {"tool": "featureToPoint", "params": {"input": "Municipios", "output": "Viviendas rurales sin servicio"}}
# Los polígonos ya no hacen falta: fuera, para que el proyecto pese menos
layer.remove {"layer": "Municipios"}

# La máscara: el rectángulo del tinte menos los departamentos = todo lo que no es Colombia
gp.run {"tool": "erase", "params": {"input": "Tinte ámbar", "erase": "Departamentos", "output": "Fuera de Colombia"}}
layer.color {"layer": "Fuera de Colombia", "color": "#07090FE6"}
layer.outline {"layer": "Fuera de Colombia", "color": "#07090F00", "width": 0}

# Las luces de Colombia a todo su brillo: el tinte en multiply, en ámbar claro (#FFC46B). Se probó
# opacarlas (#A87F45 y #D4A15A, 2026-11-15) y el mapa perdía fuerza; lo que competía con los círculos
# eran los vecinos, y eso lo resuelve la máscara. No se baja la opacidad de las teselas: el fondo de
# blank es blanco y las luces se volverían grises
layer.color {"layer": "Tinte ámbar", "color": "#FFC46B"}
layer.outline {"layer": "Tinte ámbar", "color": "#FFC46B00", "width": 0}
layer.blend {"layer": "Tinte ámbar", "mode": "multiply"}

# Departamentos: sin relleno, una línea gris tenue para ubicarse, sin competir con las luces
layer.noFill {"layer": "Departamentos"}
layer.outline {"layer": "Departamentos", "color": "#3A4354", "width": 0.4}

# Las viviendas rurales sin servicio, encendidas: círculos cian semitransparentes para que los encimados
# se lean, con borde claro. Área proporcional a las viviendas: diámetro = 0,6 + 0,15·√v puntos de
# símbolo (1/96 de pulgada, 0,265 mm). Uribia (49 180) da 33,9 pt, unos 9 mm; 10 000, 15,6 pt (4,1 mm);
# 1 000, 5,3 pt (1,4 mm); 100, 2,1 pt (0,56 mm)
layer.marker {"layer": "Viviendas rurales sin servicio", "shape": "circle", "size": 4}
layer.color {"layer": "Viviendas rurales sin servicio", "color": "#00E5FF73"}
layer.outline {"layer": "Viviendas rurales sin servicio", "color": "#7DF3FF", "width": 0.6}
layer.symbolBy {"layer": "Viviendas rurales sin servicio", "size": "0.6 + 0.15 * sqrt(vssrural)"}

# Nombres solo para los seis más grandes (18 000 o más): Uribia 49 180, Tumaco 32 155, Manaure 24 051,
# Valledupar 22 717, Cartagena 20 572 y Cumaribo 18 335. La UPME escribe "Cartagena De Indias" y "San
# Andrés De Tumaco": van con su nombre corto, que además tapa menos círculos. Comprobado en la primera
# prueba: la etiqueta vacía no dibuja nada
labels.by {"layer": "Viviendas rurales sin servicio", "text": "CASE WHEN vssrural < 18000 THEN '' WHEN municipio = 'Cartagena De Indias' THEN 'Cartagena' WHEN municipio = 'San Andrés De Tumaco' THEN 'Tumaco' ELSE municipio END"}
labels.style {"layer": "Viviendas rurales sin servicio", "size": 7, "color": "#E6F7FF", "halo": true, "haloColor": "#07090F"}
labels.placement {"layer": "Viviendas rurales sin servicio", "position": "right", "showAll": true}

# Orden de arriba abajo: círculos, departamentos, máscara, tinte (sobre las teselas). Cada capa nueva
# entra arriba, así que ahora es: máscara, círculos, departamentos, tinte. Una lista en layer.move
# conserva el orden relativo, así que se mueve solo la máscara, al tercer puesto (to = 2; no está ahí,
# así que no da el error de "already there")
layer.move {"layer": "Fuera de Colombia", "to": 2}
